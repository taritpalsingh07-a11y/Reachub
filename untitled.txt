  <script>
    const input = document.getElementById('nameInput');
    const container = document.getElementById('fontResults');
    const modal = document.getElementById('copyModal');
    const countdownBadge = document.getElementById('countdownBadge');

    let textToCopy = '';
    let countdownInterval = null;

    // Unicode font maps
    const smallCapsMap = {'a':'ᴀ','b':'ʙ','c':'ᴄ','d':'ᴅ','e':'ᴇ','f':'ꜰ','g':'ɢ','h':'ʜ','i':'ɪ','j':'ᴊ','k':'ᴋ','l':'ʟ','m':'ᴍ','n':'ɴ','o':'ᴏ','p':'ᴘ','q':'ǫ','r':'ʀ','s':'s','t':'ᴛ','u':'ᴜ','v':'ᴠ','w':'ᴡ','x':'x','y':'ʏ','z':'ᴢ'};
    const doubleStruckMap = {'a':'𝕒','b':'𝕓','c':'𝕔','d':'𝕕','e':'𝕖','f':'𝕗','g':'𝕘','h':'𝕙','i':'𝕚','j':'𝕛','k':'𝕜','l':'𝕝','m':'𝕞','n':'𝕟','o':'𝕠','p':'𝕡','q':'𝕢','r':'𝕣','s':'𝕤','t':'𝕥','u':'𝕦','v':'𝕧','w':'𝕨','x':'𝕩','y':'𝕪','z':'𝕫','A':'𝔸','B':'𝔹','C':'ℂ','D':'𝔻','E':'𝔼','F':'𝔽','G':'𝔾','H':'ℍ','I':'𝕀','J':'𝕁','K':'𝕂','L':'𝕃','M':'𝕄','N':'ℕ','O':'𝕆','P':'ℙ','Q':'ℚ','R':'ℝ','S':'𝕊','T':'𝕋','U':'𝕌','V':'𝕍','W':'𝕎','X':'𝕏','Y':'𝕐','Z':'ℤ'};
    const scriptMap = {'a':'𝒶','b':'𝒷','c':'𝒸','d':'𝒹','e':'ℯ','f':'𝒻','g':'ℊ','h':'𝒽','i':'𝒾','j':'𝒿','k':'𝓀','l':'𝓁','m':'𝓂','n':'𝓃','o':'ℴ','p':'𝓅','q':'𝓆','r':'𝓇','s':'𝓈','t':'𝓉','u':'𝓊','v':'𝓋','w':'𝓌','x':'𝓍','y':'𝓎','z':'𝓏','A':'𝒜','B':'ℬ','C':'𝒞','D':'𝒟','E':'ℰ','F':'ℱ','G':'𝒢','H':'ℋ','I':'ℐ','J':'𝒥','K':'𝒦','L':'ℒ','M':'ℳ','N':'𝒩','O':'𝒪','P':'𝒫','Q':'𝒢','R':'ℛ','S':'𝒮','T':'𝒯','U':'𝒰','V':'𝒱','W':'𝒲','X':'𝒳','Y':'𝒴','Z':'𝒵'};

    function transform(str, map) {
      return str.split('').map(ch => map[ch] || map[ch.toLowerCase()] || ch).join('');
    }

    function generateStyles(base) {
      if (!base.trim()) base = "ReachHub";
      return [
        transform(base, smallCapsMap),
        transform(base, doubleStruckMap),
        transform(base, scriptMap),
        `꧁༒${base}༒꧂`,
        `亗 ${base} 亗`,
        `メ${base}メ`,
        `▄︻デ${base}══━一`,
        `々${base}々`,
        `★${base}★`,
        `⚡${base}⚡`,
        `〖${base}〗`,
        `༺${base}༻`,
        `『V I P』${base}`,
        `〆${base}〆`,
        `★彡[${base}]彡★`,
        `【${base}】`
      ];
    }

    function render() {
      const val = input.value;
      const styles = generateStyles(val);
      container.innerHTML = '';

      styles.forEach((item) => {
        const card = document.createElement('div');
        card.className = 'font-card';

        const textSpan = document.createElement('span');
        textSpan.className = 'font-text';
        textSpan.textContent = item;

        const btn = document.createElement('button');
        btn.className = 'copy-btn';
        btn.textContent = 'Copy';
        btn.onclick = () => startCopyFlow(item, btn);

        card.appendChild(textSpan);
        card.appendChild(btn);
        container.appendChild(card);
      });
    }

    function startCopyFlow(text, btnElement) {
      textToCopy = text;
      let timeLeft = 5;
      countdownBadge.textContent = `Ready in ${timeLeft}s`;
      modal.style.display = 'flex';

      if (countdownInterval) clearInterval(countdownInterval);

      countdownInterval = setInterval(() => {
        timeLeft--;
        if (timeLeft > 0) {
          countdownBadge.textContent = `Ready in ${timeLeft}s`;
        } else {
          clearInterval(countdownInterval);
          navigator.clipboard.writeText(textToCopy).then(() => {
            modal.style.display = 'none';
            btnElement.textContent = 'Copied! ✓';
            btnElement.style.background = '#10b981';
            btnElement.style.color = '#fff';
            setTimeout(() => {
              btnElement.textContent = 'Copy';
              btnElement.style.background = '';
              btnElement.style.color = '';
            }, 2000);
          });
        }
      }, 1000);
    }

    input.addEventListener('input', render);
    render();
  </script>
